## Task 2: Variable Files and Precedence

```

# terrform.tfvars

region = "eu-north-1"

vpc_cidr = "10.0.0.0/16"

subnet-cidr = "10.0.1.0/26"

instance_type = "t3.medium"

ami-id = "ami-0aba19e56f3eaec05"

allowed-ports = [22, 80, 443]

project-name = "terraweek"

```
**This has higher precedence than variables.tf**


```
#prod.tfvars

vpc_cidr = "10.0.0.0/24"

subnet-cidr = "10.0.1.0/28"

instance_type = "t3.large"

ami-id = "ami-0aba19e56f3eaec05"

allowed-ports = [22, 80, 443]

```

**for using this you ahve to run terraform plan -var-file="prod.tfvars"**

----------------------------------

**to override every var file use -var from CLI, that has highest precendence**

`terraform plan -var="instance_type=t2.nano"`

## Task 3: Add Outputs

```
output "vpc-id" {
    value = aws_vpc.main.id
  
}

output "subnet-id" {
  value = aws_subnet.public1.id
}

output "instance-id" {
    value = aws_instance.my-instance.id
  
}

output "public-dns" {
    value = aws_instance.my-instance.public_dns
}

output "private-ip" {
    value = aws_instance.my-instance.private_ip
  
}

output "security-group-id" {
    value = aws_security_group.security_group.id
  
}
```

```
terraform apply

# After apply, you can also run:
terraform output                          # Show all outputs
terraform output instance_public_ip       # Show a specific output
terraform output -json                    # JSON format for scripting
```

## Task 4: Use Data Sources

```
data "aws_ami" "ubuntu" {
    most_recent = true 
    owners = ["099720109477"]

    filter {
        name = "name"
        values = ["ubuntu/images/hvm-ssd/ubuntu-jammy-22.04-amd64-server-*"]
    }

    filter {
        name = "virtualization-type"
        values = ["hvm"]
    }

    filter {
        name = "root-device-type"
        values = ["ebs"]
    }

}

use in ec2 block s -
`ami = data.aws_ami.ubuntu.id`

data "aws_availability_zones" "zones" {  # it should be aws_availabilty_zones
    state = "available"

    filter {
      name = "opt-in-status"
      values = ["opt-in-not-required"]
    }
  
}

# add this in subnet block - availability_zone = data.aws_availability_zones.zones.names[0]. make sure to add names as list in the end.


data "aws_ec2_instance_types" "ins_type" {
    filter {
        name = "processor-info.supported-architecture"
        values = ["x86_64"]

    }  
    filter {
        name = "vcpu-info.default-vcpus"
        values = ["2"]
    }

    filter {
        name = "memory-info.size-in-mib"
        values = ["4096"]
    }
    filter {
    name   = "instance-type"
    values = ["c7*"] 
  }
}

use in instance block as -

instance_type = data.aws_ec2_instance_types.ins_type.instance_types[2]

# amke sure to add instance_types in the end with list.
```

## Task 6: Built-in Functions and Conditional Expressions

**Practice these in terraform console:**

terraform console

**String functions:**

upper("terraweek") -> "TERRAWEEK"
join("-", ["terra", "week", "2026"]) -> "terra-week-2026"
format("arn:aws:s3:::%s", "my-bucket")

**Collection functions:**

length(["a", "b", "c"]) -> 3
lookup({dev = "t2.micro", prod = "t3.small"}, "dev") -> "t2.micro"
toset(["a", "b", "a"]) -> removes duplicates

**Networking function:**

cidrsubnet("10.0.0.0/16", 8, 1) -> "10.0.1.0/24"

Conditional expression -- add this to your config:

instance_type = var.environment == "prod" ? "t3.small" : "t2.micro"

Apply with environment = "prod" and verify the instance type changes.
